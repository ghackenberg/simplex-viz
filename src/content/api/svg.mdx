## Mounting in Vanilla TypeScript, Vue, Svelte, or Angular

`createSimplexSvg` accepts either a CSS selector (`'#atlas'`) or an `HTMLElement` reference and returns a `SimplexSvgController` whose `.destroy()` method cleans up all DOM listeners and `ResizeObserver` instances:

```ts
import {
  createSimplexSvg,
  createSubspaceZoomStrategy,
  GradientTriangleSvgView,
  ClusterTintedEdgeSvgView,
} from '@simplex-viz/svg';

const controller = createSimplexSvg('#atlas', {
  items,
  preset: 'glyphs',
  theme: 'dark',
  triangleView: new GradientTriangleSvgView({ baseOpacity: 0.16 }),
  edgeView: new ClusterTintedEdgeSvgView({ strokeWidth: 1.35 }),
  zoomStrategy: createSubspaceZoomStrategy(),
  showZoomBreadcrumbs: true,
  itemFilter: (item) => item.tags.includes('geometry'),
  itemFilterMode: 'dim',
  ariaLabel: 'Interactive knowledge simplex atlas',
  keyboardNavigation: true,
  tooltip: {
    render(target, container) {
      if (target.kind !== 'content') return false;
      const title = document.createElement('strong');
      title.textContent = target.item.label;
      container.replaceChildren(title);
    },
  },
  onLegendChange: (legend) => {
    console.log('Legend entries:', legend.entries.length, 'visible:', legend.visibleItems);
  },
});

// Coalesce multi-property runtime updates into a single recomputation:
controller.setOptions({
  theme: 'light',
  itemFilterMode: 'hide',
  highlightedIds: ['item-1', 'item-4'],
});

// Programmatic linked-view navigation, legend snapshot, multi-selection, deep-linking & export:
const legend = controller.getLegendModel();
controller.zoomToItem('item-1', { zoom: 2.2, maxDrillSteps: 4 });
controller.setSelectedIds(['item-1', 'item-4']);
const snapshot = controller.getViewState();
controller.restoreViewState(snapshot);
const svgMarkup = controller.exportSvg({ padding: 24, title: 'Knowledge Atlas' });
const pngBlob = await controller.exportPng({ scale: 2, padding: 24 });

// Clean up on unmount:
// controller.destroy();
```

## Headless Zero-DOM SSR (`@simplex-viz/svg/ssr`)

Import `renderSimplexSvgString` from `@simplex-viz/svg/ssr` in Node.js, Edge runtimes, Astro/Next.js builds, or headless PNG pipelines (such as `@resvg/resvg-js`) without bundling browser controller code or running Puppeteer:

```ts
import { renderSimplexSvgString } from '@simplex-viz/svg/ssr';

const svg = renderSimplexSvgString(items, {
  width: 1200,
  height: 720,
  preset: 'badges',
  theme: 'dark',
  padding: 24,
});
```

## Custom `SvgNodeView` Contract

Implement `SvgNodeView<TVisual, TRenderState>` when building high-throughput custom SVG glyphs that update imperatively at 60 fps without virtual-DOM overhead:

```ts
import type { SvgNodeView, ContentNodeRenderState } from '@simplex-viz/svg';

export const pillNodeView: SvgNodeView<
  { label: string; colorHex: string },
  ContentNodeRenderState<{ label: string; colorHex: string }, MyItem>
> = {
  createElement(doc) {
    const g = doc.createElementNS('http://www.w3.org/2000/svg', 'g');
    const rect = doc.createElementNS('http://www.w3.org/2000/svg', 'rect');
    const text = doc.createElementNS('http://www.w3.org/2000/svg', 'text');
    g.append(rect, text);
    return g;
  },
  updateElement(g, visual, state) {
    const [rect, text] = g.children as unknown as [SVGRectElement, SVGTextElement];
    rect.setAttribute('x', '-42');
    rect.setAttribute('y', '-12');
    rect.setAttribute('width', '84');
    rect.setAttribute('height', '24');
    rect.setAttribute('rx', '6');
    rect.setAttribute('fill', visual.colorHex);
    rect.setAttribute('opacity', state.interaction.isDimmed ? '0.28' : '1');
    text.textContent = visual.label;
  },
};
```
